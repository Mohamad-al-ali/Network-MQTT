TYPE_CONNECT = 0x10
TYPE_CONNACK = 0x20
TYPE_DISCONNECT = 0xe0
TYPE_PUBLISH = 0x30
TYPE_SUBSCRIBE = 0x80
TYPE_SUBACK = 0x90
PROTOCOLE_NAME = "MQTT"
PROTOCOLE_VERSION = 0x4
QOS_DEFAULT = 0

def create_mqtt_publish_msg(topic, value, retain=False):
    #control header
    if retain:
        flag=bytes('1','ascii')
    else :
        flag=bytes('0','ascii')

    #longueur totale
    lenmsg=(len(topic)+2+len(value)).to_bytes(1,byteorder='big')

    #longueur du topic
    lentopic=len(topic).to_bytes(2,byteorder='big')

    #return la reponse encodé
    return flag+lenmsg+lentopic+bytes(topic+value,'ascii')

def create_mqtt_connect_msg(client_id):
    msg_mqtt_flags = TYPE_CONNECT.to_bytes(1,byteorder='big')
    msg_mqtt_protocoleName= PROTOCOLE_NAME.encode("ascii")
    msg_mqtt_protocoleLenght=len( msg_mqtt_protocoleName).to_bytes(2,byteorder='big')
    msg_mqtt_level= PROTOCOLE_VERSION.to_bytes(1,byteorder='big')
    msg_mqtt_id=client_id.encode("ascii")
    msg_mqtt_id_lenght=len(msg_mqtt_id).to_bytes(2,byteorder='big')
    msg_mqtt_connect=(2).to_bytes(1,byteorder='big')
    msg_mqtt_keepalive=(60).to_bytes(1,byteorder='big')
    msg_mqtt= msg_mqtt_protocoleLenght + msg_mqtt_protocoleName + msg_mqtt_level + msg_mqtt_connect + msg_mqtt_keepalive + msg_mqtt_id_lenght + msg_mqtt_id
    msg_mqtt_lenght=len(msg_mqtt).to_bytes(1,byteorder='big')
    
    return msg_mqtt_flags + msg_mqtt_lenght + msg_mqtt
    
def create_mqtt_connack_msg():
    msg_mqtt_flags = TYPE_CONNACK.to_bytes(1,byteorder='big')
    msg_flag = (0x00).to_bytes(1, byteorder = 'big')
    msg_mqtt_code=(0).to_bytes(1,byteorder='big')
    msg=msg_flag + msg_mqtt_flags +msg_mqtt_code
    msg_mqtt_lenght=len(msg).to_bytes(1,byteorder='big')
  
    return msg_mqtt_flags + msg_mqtt_lenght + msg_flag + msg_mqtt_code

def create_mqtt_disconnect_msg():
    msg_mqtt_flags=TYPE_DISCONNECT.to_bytes(1,byteorder='big')
    msg_mqtt_lenght=(0).to_bytes(1,byteorder='big')
    return msg_mqtt_flags + msg_mqtt_lenght
    
def create_mqtt_subscribe_msg(topic, packet_id):
    msg_mqtt_flags=TYPE_SUBSCRIBE.to_bytes(1,byteorder='big')
    msg_mqtt_id=packet_id.encode("ascii")
    msg_topic = topic.encode("ascii")
    msg_topic_length = len(msg_topic).to_bytes(2, byteorder='big')
    msg=  msg_mqtt_id + msg_topic_length + msg_topic
    msg_mqtt_lenght=len(msg).to_bytes(1,byteorder='big')
    return msg_mqtt_flags + msg_mqtt_lenght + msg_mqtt_id + msg_topic_length + msg_topic
    
def create_mqtt_suback_msg(topic, packet_id):
    msg_mqtt_flags=TYPE_SUBACK.to_bytes(1,byteorder='big')
    msg_mqtt_id=packet_id.encode("ascii")
    msg_mqtt_requested_QoS= (0).to_bytes(1,byteorder='big')
    msg=msg_mqtt_id + msg_mqtt_requested_QoS
    msg_mqtt_lenght=len(msg).to_bytes(1,byteorder='big')
    return msg_mqtt_flags + msg_mqtt_lenght + msg
